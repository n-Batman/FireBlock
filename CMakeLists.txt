cmake_minimum_required(VERSION 3.24)
project(FireBlock VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "FireBlock currently uses Windows WinHTTP and BCrypt. Build on Windows.")
endif()
add_library(fireblock_analysis_worker src/analysis_worker.cpp)
target_include_directories(fireblock_analysis_worker PUBLIC include)
target_compile_definitions(fireblock_analysis_worker PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(fireblock_analyze src/analysis_main.cpp)
target_link_libraries(fireblock_analyze PRIVATE fireblock_analysis_worker)
target_compile_definitions(fireblock_analyze PRIVATE FIREBLOCK_ANALYSIS_ENTRY="${CMAKE_CURRENT_SOURCE_DIR}/python/worker_entry.py")
find_package(nlohmann_json 3.12.0 QUIET)
if(NOT nlohmann_json_FOUND)
  include(FetchContent)
  FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE TIMEOUT 45 INACTIVITY_TIMEOUT 15)
  set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(json)
endif()
add_library(fireblock src/controller.cpp src/windows_http.cpp)
target_include_directories(fireblock PUBLIC include)
target_link_libraries(fireblock PUBLIC nlohmann_json::nlohmann_json PRIVATE winhttp bcrypt)
target_compile_definitions(fireblock PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(fireblock PRIVATE /W4 /utf-8)
endif()
add_executable(fireblock_cli src/main.cpp)
set_target_properties(fireblock_cli PROPERTIES OUTPUT_NAME fireblock)
target_link_libraries(fireblock_cli PRIVATE fireblock)
add_library(fireblock_ingestion src/ingest.cpp)
target_link_libraries(fireblock_ingestion PUBLIC fireblock)
target_compile_definitions(fireblock_ingestion PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(fireblock_ingestion PRIVATE /W4 /utf-8)
endif()
add_executable(fireblock_ingest src/ingest_main.cpp)
target_link_libraries(fireblock_ingest PRIVATE fireblock_ingestion)
add_library(fireblock_raw_data src/raw_data.cpp)
target_link_libraries(fireblock_raw_data PUBLIC fireblock_ingestion)
target_compile_definitions(fireblock_raw_data PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(fireblock_raw_data PRIVATE /W4 /utf-8)
endif()
add_executable(fireblock_pipeline src/pipeline_main.cpp)
target_link_libraries(fireblock_pipeline PRIVATE fireblock_raw_data)
add_library(fireblock_csv src/csv.cpp)
target_include_directories(fireblock_csv PUBLIC include)
add_library(fireblock_parsing src/parsing.cpp)
target_link_libraries(fireblock_parsing PUBLIC fireblock fireblock_csv)
target_compile_definitions(fireblock_parsing PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(fireblock_parsing PRIVATE /W4 /utf-8)
endif()
add_executable(fireblock_parse src/parsing_main.cpp)
target_link_libraries(fireblock_parse PRIVATE fireblock_parsing fireblock_ingestion)
include(CTest)
if(BUILD_TESTING)
  add_test(NAME analysis_help COMMAND fireblock_analyze --help)
  add_executable(fireblock_tests tests/controller_tests.cpp)
  target_link_libraries(fireblock_tests PRIVATE fireblock)
  add_test(NAME controller COMMAND fireblock_tests)
  add_executable(fireblock_retry_tests tests/retry_tests.cpp)
  target_link_libraries(fireblock_retry_tests PRIVATE fireblock)
  add_test(NAME storage_retries COMMAND fireblock_retry_tests)
  add_test(NAME cli_help COMMAND fireblock_cli help)
  add_executable(fireblock_ingest_tests tests/ingest_tests.cpp)
  target_link_libraries(fireblock_ingest_tests PRIVATE fireblock_ingestion)
  add_test(NAME ingestion COMMAND fireblock_ingest_tests)
  add_test(NAME ingestion_help COMMAND fireblock_ingest --help)
  add_test(NAME ingestion_config COMMAND fireblock_ingest ${CMAKE_CURRENT_SOURCE_DIR}/config/ingest.example.json --validate)
  add_executable(fireblock_parsing_tests tests/parsing_tests.cpp)
  target_link_libraries(fireblock_parsing_tests PRIVATE fireblock_parsing)
  add_test(NAME parsing COMMAND fireblock_parsing_tests)
  add_test(NAME parsing_help COMMAND fireblock_parse --help)
  add_test(NAME parsing_config COMMAND fireblock_parse ${CMAKE_CURRENT_SOURCE_DIR}/config/parsing.example.json --validate)
  add_executable(fireblock_csv_tests tests/csv_tests.cpp)
  target_link_libraries(fireblock_csv_tests PRIVATE fireblock_csv)
  add_test(NAME csv COMMAND fireblock_csv_tests)
  add_executable(fireblock_pipeline_tests tests/pipeline_tests.cpp)
  target_link_libraries(fireblock_pipeline_tests PRIVATE fireblock_raw_data)
  add_test(NAME raw_pipeline COMMAND fireblock_pipeline_tests)
  add_test(NAME raw_pipeline_help COMMAND fireblock_pipeline --help)
  add_executable(fireblock_dedupe_tests tests/dedupe_tests.cpp)
  target_link_libraries(fireblock_dedupe_tests PRIVATE fireblock_raw_data)
  add_test(NAME satellite_dedupe COMMAND fireblock_dedupe_tests)
